# Необходимо реализовать функцию-обёртку, аналогичную jQuery,

Необходимо реализовать функцию-обёртку, аналогичную jQuery, 
  которая позволяет работать с элементами DOM в виде цепочки вызовов.
  Она должна поддерживать методы для обработки событий, изменения HTML-содержимого и изменения CSS-стилей.

  $("button").on("click", () => alert("Button clicked!"));

  $("div")
      .html("New Content")
      .css("background-color", "blue")
      .on("mouseover", () => console.log("Mouse over!"));
